---
title: List Projects
description: Retrieve a paginated list of projects in a workspace
---

`GET /public/workspace/{workspaceId}/projects`

Returns a paginated list of projects in the specified workspace.

## Path parameters

<ParamField path="workspaceId" type="string" required>
  The workspace ID
</ParamField>

## Query parameters

<ParamField query="page" type="integer" default="1">
  Page number (1-indexed)
</ParamField>
<ParamField query="per_page" type="integer" default="5">
  Items per page (min: 1, max: 10)
</ParamField>
<ParamField query="sort_by" type="string" default="created_at">
  Sort field. One of: `created_at`, `project_title`
</ParamField>
<ParamField query="sort_direction" type="string" default="desc">
  Sort direction. One of: `asc`, `desc`
</ParamField>

## Response

<ResponseField name="message" type="string">
  Status message
</ResponseField>
<ResponseField name="data" type="object">
  <Expandable title="data">
    <ResponseField name="workspaceId" type="string">
      Workspace ID
    </ResponseField>
    <ResponseField name="projects" type="array">
      <Expandable title="project item">
        <ResponseField name="projectId" type="string">
          Project ID
        </ResponseField>
        <ResponseField name="projectTitle" type="string">
          Project title
        </ResponseField>
        <ResponseField name="isDefault" type="boolean">
          Whether this is the default project
        </ResponseField>
        <ResponseField name="recordingRetentionDays" type="integer">
          Recording retention period in days
        </ResponseField>
        <ResponseField name="createdAt" type="string">
          ISO 8601 timestamp
        </ResponseField>
        <ResponseField name="updatedAt" type="string">
          ISO 8601 timestamp
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="pagination" type="object">
      [Pagination metadata](/api/pagination#pagination-metadata)
    </ResponseField>
  </Expandable>
</ResponseField>

## Example

<CodeGroup>
```bash curl
curl "https://api.userplane.com/api/v1/public/workspace/ws_abc123/projects?page=1&per_page=5" \
  -H "Authorization: Bearer up_your_api_key"
```

```javascript JavaScript
const response = await fetch(
  'https://api.userplane.com/api/v1/public/workspace/ws_abc123/projects?page=1&per_page=5',
  { headers: { Authorization: 'Bearer up_your_api_key' } }
);
const { data } = await response.json();
```
