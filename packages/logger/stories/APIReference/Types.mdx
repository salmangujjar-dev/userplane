# Types Reference

All TypeScript types exported by `@userplane/logger`.

## Log Events

These are the raw event types dispatched by interceptors via `CustomEvent`.

### ConsoleLogEvent

```typescript
interface ConsoleLogEvent {
  type: 'console';
  level: 'log' | 'info' | 'warn' | 'error' | 'debug' | 'trace';
  args: any[];
  trace?: {
    first: string;
    lines: string[];
  };
  repeatCount?: number;
  subtype?: 'securitypolicyviolation';
  violatedDirective?: string;
  blockedURI?: string;
  effectiveDirective?: string;
  disposition?: string;
  meta?: {
    truncated?: {
      trace?: boolean;
      args?: boolean;
    };
  };
}
```

### NetworkLogEvent

```typescript
interface NetworkLogEvent {
  type: 'network';

  responseHeaders: Record<string, string>;
  requestBody?: any;
  responseBody?: string;
  performanceData?: PerformanceData;
  timestamp: number;
  initiatorType: 'fetch' | 'xmlhttprequest';
```
```typescript
}
```

## Processed Entries

Normalized entries produced by the LogProcessor for storage/dispatch.

### ProcessedNetworkEntry

```typescript
interface ProcessedNetworkEntry {
  url: string;
  method: string;
  status: number;
  statusText: string;
  requestHeaders: Record<string, string>;
  responseHeaders: Record<string, string>;
  requestBody?: any;
  responseBody?: string;
  protocol?: string;
  duration?: number;
  performanceData?: PerformanceData;
  timestamp: number;
  contentType?: string;
}
```

### ProcessedConsoleEntry

```typescript
interface ProcessedConsoleEntry {
```

## Dispatch Types

### SavePayload

```typescript
interface SavePayload {
  type: LogStoreType;
  entry: ProcessedNetworkEntry | ProcessedConsoleEntry | ProcessedUserActionEntry;
  tabId: string;
  tabState: string;
  timestamp: number;
  focused: boolean;
}
```

### BatchSavePayload

```typescript
interface BatchSavePayload {
  entries: Array<{
    type: LogStoreType;
    entry: ProcessedNetworkEntry | ProcessedConsoleEntry | ProcessedUserActionEntry;
  }>;
  tabId: string;
  tabState: string;
  timestamp: number;
  focused: boolean;
}
```

### DispatchStrategy

```typescript
interface DispatchStrategy {
  emit(entry: ProcessedLogEntry): void;
  start(): void;
  pause(): void;
  stop(): void;
  destroy(): void;
}
```

### DispatchContext

```typescript
interface DispatchContext {
  tabId: string;
  serverNow: () => number;
  saveFunction: (payload: SavePayload | BatchSavePayload) => Promise<void>;
}
```
